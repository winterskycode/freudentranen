# mod-base.bin made from the player's files (modbase.h): a static library and its command line. Needs zstd: the caller's
# target, or the system's when built alone.
#   PC:     cmake -G Ninja -S port/android/setup/modbase -B <dir> && ninja -C <dir>
#   -DTOKM_SANITIZE=ON: ASan and UBSan.
cmake_minimum_required(VERSION 3.22)
project(tok_modbase C)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS ON)
option(TOKM_SANITIZE "build with ASan and UBSan" OFF)
option(TOKM_TESTS "build the damage test" OFF)
if(TOKM_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=address,undefined)
endif()
if(NOT TARGET tokcontainer)
  add_subdirectory(../container container)
endif()
add_library(tokmodbase STATIC modbase.c)
target_include_directories(tokmodbase PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} PRIVATE ../container ${TOKM_ZSTD_INCLUDE})
target_compile_options(tokmodbase PRIVATE -Wall -Wextra -Wshadow -Wconversion)
set_target_properties(tokmodbase PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(tokmodbase PUBLIC tokcontainer)
if(NOT ANDROID)
  find_package(Threads REQUIRED)
  add_executable(modbase_cli modbase_cli.c)
  target_compile_options(modbase_cli PRIVATE -Wall -Wextra)
  target_link_libraries(modbase_cli PRIVATE tokmodbase zstd Threads::Threads)
  if(TOKM_TESTS)
    add_executable(modbase_damage tests/damage.c)
    target_include_directories(modbase_damage PRIVATE ../container)
    target_link_libraries(modbase_damage PRIVATE tokcontainer zstd Threads::Threads)
  endif()
endif()
