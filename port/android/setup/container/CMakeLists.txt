# The reader for the player's own game files (container.h): a static library, its command line and tests. Stands alone;
# nothing else in the project builds or links it yet. No third-party code.
#   PC:     cmake -G Ninja -S port/android/setup/container -B <dir> && ninja -C <dir>
#   Device: build_android.sh <dir>
#   -DTOKC_SANITIZE=ON: ASan and UBSan. -DTOKC_TSAN=ON: ThreadSanitizer. -DTOKC_FUZZER=ON: also the libFuzzer target (clang).
#   -DTOKC_NO_HW=ON: the portable AES only, to test it on real files.
cmake_minimum_required(VERSION 3.22)
project(tok_container C)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS ON)
option(TOKC_SANITIZE "build with ASan and UBSan" OFF)
option(TOKC_TSAN "build with ThreadSanitizer" OFF)
option(TOKC_FUZZER "build the libFuzzer target" OFF)
option(TOKC_NO_HW "portable AES only" OFF)
if(TOKC_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=address,undefined)
endif()
if(TOKC_TSAN)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=thread)
endif()
if(TOKC_FUZZER)
  add_compile_options(-fsanitize=fuzzer-no-link)
endif()

if(TOKC_NO_HW)
  add_compile_definitions(TOKC_NO_HW)
endif()
add_library(tokcontainer STATIC aes.c aes_hw.c sha256.c keys.c package.c bktr.c game.c romfs_index.c)
target_include_directories(tokcontainer PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(tokcontainer PRIVATE -Wall -Wextra -Wshadow -Wconversion)
set_target_properties(tokcontainer PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_executable(container_cli container_cli.c tests/selftest.c)
target_compile_options(container_cli PRIVATE -Wall -Wextra)
target_link_libraries(container_cli PRIVATE tokcontainer)
find_package(Threads REQUIRED)
target_link_libraries(container_cli PRIVATE Threads::Threads)

# Tests (regress.sh runs them). Made-up files only.
if(NOT ANDROID)
  add_library(tokc_synth STATIC tests/synth.c)
  target_link_libraries(tokc_synth PUBLIC tokcontainer)
  add_executable(test_synth tests/test_synth.c tests/selftest.c)
  target_link_libraries(test_synth PRIVATE tokc_synth Threads::Threads)
  add_executable(fuzz_malformed tests/fuzz_malformed.c tests/fuzz_common.c)
  target_link_libraries(fuzz_malformed PRIVATE tokc_synth)
  add_executable(key_check tests/key_check.c)
  target_link_libraries(key_check PRIVATE tokcontainer)
  if(TOKC_FUZZER)
    add_executable(fuzz_libfuzzer tests/fuzz_libfuzzer.c tests/fuzz_common.c)
    target_link_libraries(fuzz_libfuzzer PRIVATE tokc_synth)
    target_link_options(fuzz_libfuzzer PRIVATE -fsanitize=fuzzer)
  endif()
endif()
